    <script>
        const API_URL = "https://script.google.com/macros/s/AKfycbxXntnyiuk4NaQgSfjMu3eZSum-nHIOh4oPM8XMcthn55ExTAnq1AUXk3GLzVFE2Kq7/exec";

        let currentSelectedFolderId = "";

        let sessionUser = JSON.parse(localStorage.getItem('saobay_user')) || null;
        let currentUser = {
            id: sessionUser ? sessionUser.id : (localStorage.getItem('user_id') || 'GV_GUEST'),
            name: sessionUser ? sessionUser.name : (localStorage.getItem('user_name') || 'Giáo viên'),

        let scoreConfig = {

        function initTeacherHub() {
            if (currentUser.role === 'admin') {
                const adminBtn = document.getElementById('admin-settings-btn');
                if (adminBtn) adminBtn.classList.remove('hidden');
            }

            loadFolderTreeData();
            updateUserUI();
            updateScorePreview();
        }

        function updateUserUI() {
            document.getElementById('user-display-name').innerText = currentUser.name;
            document.getElementById('user-total-score').innerText = currentUser.score;

            let levelTag = document.getElementById('user-level-tag');
            let score = currentUser.score;

            if (score >= 1000) {
                levelTag.innerHTML = `<i class="fa-solid fa-tree text-emerald-400 mr-1"></i>Cây Cổ Thụ`;
            } else if (score >= 500) {
                levelTag.innerHTML = `<i class="fa-solid fa-leaf text-green-400 mr-1"></i>Lá Xanh`;
            } else if (score >= 200) {
                levelTag.innerHTML = `<i class="fa-solid fa-plant-wilt text-lime-300 mr-1"></i>Chồi Non`;
            } else {
                levelTag.innerHTML = `<i class="fa-solid fa-seedling text-amber-300 mr-1"></i>Mầm`;
            }
        }

        function updateScorePreview() {
            let type = document.getElementById('item-type').value;
            let preview = document.getElementById('score-reward-preview');
            let isEdit = document.getElementById('edit-mode-flag').value === 'EDIT';

            if (isEdit) {
                preview.innerText = `+${scoreConfig.edit} điểm thưởng (Cập nhật/Sửa bài)`;
                preview.className = "text-xs font-bold text-blue-600 bg-blue-50 px-3 py-1.5 rounded-lg border border-blue-200";
            } else {
                let pts = scoreConfig.knowledge;
                if (type === 'EXAM_LONG') pts = scoreConfig.examLong;
                if (type === 'EXAM_SHORT') pts = scoreConfig.examShort;
                preview.innerText = `+${pts} điểm thưởng`;
                preview.className = "text-xs font-bold text-emerald-600 bg-emerald-50 px-3 py-1.5 rounded-lg border border-emerald-200";
            }
        }

        function toggleMathKeyboard() {
            let panel = document.getElementById('math-keyboard-panel');
            panel.classList.toggle('hidden');
        }

        function insertMath(latex) {
            let textarea = document.getElementById('item-content');
            let start = textarea.selectionStart;
            let end = textarea.selectionEnd;
            let text = textarea.value;
            let insertion = ` $${latex}$ `;
            textarea.value = text.substring(0, start) + insertion + text.substring(end);
            textarea.focus();
            renderMathPreview();
        }

        function toggleSidebar() {

            else delBtn.classList.remove('hidden');
        }

        async function parseSafeResponse(response) {
            let text = await response.text();
            try { return JSON.parse(text); } 
            catch (e) { throw new Error("Phản hồi từ Server không hợp lệ."); }
        }

        // HÀM HIỂN THỊ CÂY THƯ MỤC ĐA TẦNG PHÂN CẤP LÙI LỀ CHUẨN XÁC
        async function loadFolderTreeData() {
            let container = document.getElementById('dynamic-folder-list');
            try {
                let res = await fetch(`${API_URL}?type=ThuMuc`);
                let rawFolders = await parseSafeResponse(res);
                
                if (!Array.isArray(rawFolders) || rawFolders.length === 0) {
                    container.innerHTML = '<span class="text-slate-400 text-xs italic">Chưa có thư mục con nào.</span>';
                    return;
                }

                // 1. Chuẩn hóa thuộc tính dữ liệu
                let folders = rawFolders.map(f => {
                    let id = String(f.FolderID || f.folderId || f.id || '').trim();
                    let name = String(f.FolderName || f.folderName || f.name || 'Thư mục').trim();
                    let parentId = String(f.ParentID || f.parentId || f.ParentId || '').trim();
                    return { id, name, parentId };
                });

                // 2. Hàm đệ quy dựng giao diện dạng nhánh cây
                function buildTreeHtml(pId) {
                    let children = folders.filter(f => {
                        if (pId === ROOT_FOLDER_ID) {
                            return f.parentId === ROOT_FOLDER_ID || f.parentId === 'ROOT' || !f.parentId || !folders.some(p => p.id === f.parentId);
                        }
                        return f.parentId === pId;
                    });

                    if (children.length === 0) return '';

                    let html = '<div class="space-y-1">';
                    children.forEach(f => {
                        let safeName = f.name.replace(/'/g, "\\'");
                        let subTree = buildTreeHtml(f.id);
                        let hasSub = subTree.trim() !== '';

                        html += `
                            <div class="folder-item">
                                <div class="p-1.5 rounded hover:bg-slate-100 cursor-pointer flex items-center justify-between text-slate-700 text-xs font-medium transition border-l-2 border-transparent hover:border-blue-500" onclick="selectFolder('${f.id}', '${safeName}')">
                                    <span class="truncate flex items-center">
                                        <i class="fa-solid ${hasSub ? 'fa-folder-tree text-blue-600' : 'fa-folder text-amber-400'} mr-2"></i>
                                        ${f.name}
                                    </span>
                                </div>
                                ${hasSub ? `<div class="pl-4 border-l-2 border-slate-200 ml-2 mt-0.5 space-y-1">${subTree}</div>` : ''}
                            </div>
                        `;
                    });
                    html += '</div>';
                    return html;
                }

                let treeHtml = buildTreeHtml(ROOT_FOLDER_ID);
                container.innerHTML = treeHtml || '<span class="text-slate-400 text-xs italic">Không thể hiển thị cấu trúc cây.</span>';

            } catch (err) {
                container.innerHTML = '<span class="text-rose-500 text-xs">Lỗi kết nối danh mục thư mục.</span>';
            }
        }

        async function openAddFolderModal() {
            let folderName = prompt(`Nhập tên thư mục mới (tạo trong: "${currentSelectedFolderName}"):`);
            if(!folderName) return;

            let payload = {
                type: 'CREATE_DRIVE_FOLDER',
                parentId: currentSelectedFolderId,
                folderName: folderName,
                user: currentUser.name
            };

            try {
                let res = await fetch(API_URL, {
                    method: 'POST',
                    headers: { 'Content-Type': 'text/plain;charset=utf-8' },
                    body: JSON.stringify(payload)
                });
                let result = await parseSafeResponse(res);
                if(result.status === 'success') {
                    alert('Đã tạo thư mục thành công!');
                    loadFolderTreeData();
                } else alert('Lỗi: ' + result.message);
            } catch(err) { alert('Lỗi kết nối: ' + err.message); }
        }

        async function deleteSelectedFolder() {
            if(!currentSelectedFolderId) return;
            if(!confirm(`XÓA THƯ MỤC: Tác vụ xóa này sẽ được ghi tên "${currentUser.name}" lên hệ thống nhật ký và bị trừ ${Math.abs(scoreConfig.delete)} điểm. Tiếp tục?`)) return;

            let payload = {
                type: 'DELETE_DRIVE_FOLDER',
                folderId: currentSelectedFolderId,
                user: currentUser.name,
                action: 'DELETE_FOLDER'
            };

            try {
                let res = await fetch(API_URL, {
                    method: 'POST',
                    headers: { 'Content-Type': 'text/plain;charset=utf-8' },
                    body: JSON.stringify(payload)
                });
                let result = await parseSafeResponse(res);
                if(result.status === 'success') {
                    currentUser.score += scoreConfig.delete;
                    localStorage.setItem('user_score', currentUser.score);
                    updateUserUI();
                    alert('Đã xóa thư mục và ghi nhật ký!');
                    selectFolder('root', 'Gốc (Root Sào Báy)');
                    loadFolderTreeData();
                } else alert('Lỗi xóa: ' + result.message);
            } catch(err) { alert('Lỗi xóa: ' + err.message); }
        }

        function handleWordFileUpload(event) {
            let file = event.target.files[0];
            if (!file) return;
            let reader = new FileReader();
            reader.onload = function(loadEvent) {
                mammoth.extractRawText({arrayBuffer: loadEvent.target.result})
                    .then(function(result) {
                        document.getElementById('item-content').value = result.value;
                        renderMathPreview();
                        alert('Đã nạp file Word thành công!');
                    })
                    .catch(function(err) { alert('Lỗi đọc file: ' + err.message); });
            };
            reader.readAsArrayBuffer(file);
        }

        function renderMathPreview() {
            let content = document.getElementById('item-content').value;
            let previewBox = document.getElementById('math-preview');
            if(!content.trim()) {
                previewBox.innerHTML = '<span class="text-slate-400 italic">Công thức toán học sẽ hiển thị ở đây...</span>';
                return;
            }
            let cleanedContent = content.replace(/\\\((.*?)\\\)/g, '$$$1$$').replace(/\\\\/g, '\\').replace(/\n/g, '<br>');
            previewBox.innerHTML = cleanedContent;
            if (window.MathJax) {
                MathJax.typesetPromise([previewBox]).catch((err) => console.log(err));
            }
        }

        function insertTag(tag) {
            let textarea = document.getElementById('item-content');
            let start = textarea.selectionStart;
            let end = textarea.selectionEnd;
            let text = textarea.value;
            let insertion = `\n<${tag}>\n<Câu> 1: Nội dung câu hỏi $\\frac{a}{b}$\n<ĐA Câu 1> A\n<LG Câu 1> Lời giải chi tiết...\n</${tag}>\n`;
            textarea.value = text.substring(0, start) + insertion + text.substring(end);
            textarea.focus();
            renderMathPreview();
        }

        async function handleFormSubmit(e) {
            e.preventDefault();
            let isEdit = document.getElementById('edit-mode-flag').value === 'EDIT';
            let itemType = document.getElementById('item-type').value;

            let pointsEarned = scoreConfig.edit;
            if (!isEdit) {
                if (itemType === 'KNOWLEDGE') pointsEarned = scoreConfig.knowledge;
                else if (itemType === 'EXAM_LONG') pointsEarned = scoreConfig.examLong;
                else pointsEarned = scoreConfig.examShort;
            }

            let btn = document.getElementById('submit-btn');
            btn.disabled = true;
            btn.innerHTML = `<i class="fa-solid fa-spinner fa-spin mr-2"></i> Đang ghi nhận lên hệ thống...`;

            let payload = {
                type: 'SAVE_LESSON_CONTENT',
                actionType: isEdit ? 'EDIT_CONTENT' : 'CREATE_CONTENT',
                itemId: document.getElementById('editing-item-id').value,
                author: currentUser.name,
                authorId: currentUser.id,
                timestamp: new Date().toLocaleString(),
                folderId: currentSelectedFolderId, 
                clazz: document.getElementById('item-class').value,
                subject: document.getElementById('item-subject').value,
                title: document.getElementById('item-title').value,
                duration: document.getElementById('item-duration').value || '45',
                contentType: itemType,
                content: document.getElementById('item-content').value,
                pointsAwarded: pointsEarned
            };

            try {
                let res = await fetch(API_URL, {
                    method: 'POST',
                    headers: { 'Content-Type': 'text/plain;charset=utf-8' },
                    body: JSON.stringify(payload)
                });
                let result = await parseSafeResponse(res);

                if(result.status === 'success') {
                    currentUser.score += pointsEarned;
                    localStorage.setItem('user_score', currentUser.score);
                    updateUserUI();

                    alert(`Thành công! Thầy/Cô đã nhận được +${pointsEarned} điểm thưởng đóng góp.`);
                    resetFormToCreate();
                } else {
                    alert('Lỗi từ Server: ' + result.message);
                }
            } catch (err) {
                alert('Lỗi kết nối API: ' + err.message);
            } finally {
                btn.disabled = false;
                btn.innerHTML = `<i class="fa-solid fa-cloud-arrow-up mr-2"></i> Lưu & Đẩy Lên Hệ Thống`;
            }
        }
