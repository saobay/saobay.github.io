Bước 2: Tạo file .dotm chuẩn trên Microsoft Word
Vì file .dotm cần được tạo trực tiếp từ Microsoft Word để hệ thống hiểu đây là một Template Macro, bạn làm như sau:

Mở phần mềm Microsoft Word trên máy tính lên.

Chọn tạo một Blank Document (Tài liệu trắng).

Nhấn tổ hợp phím Alt + F11 để mở cửa sổ lập trình VBA.

Trên thanh menu, chọn Insert > Module. Một khung cửa sổ trắng hiện ra, bạn dán toàn bộ đoạn code tự động cập nhật và code chuyển đổi LaTeX sang MathType vào đây:

' --- CODE CHUYỂN ĐỔI LATEX SANG MATH TYPE ---
Sub ConvertLaTeXToMathType()
    Dim rng As Range
    Set rng = Selection.Range
    
    If rng.Text = "" Then
        MsgBox "Vui lòng bôi đen đoạn văn bản chứa công thức LaTeX cần chuyển đổi!", vbExclamation, "Thông báo"
        Exit Sub
    End If
    
    With rng.Find
        .ClearFormatting
        .Replacement.ClearFormatting
        .Text = "$(*)$"
        .MatchWildcards = True
        .Forward = True
        .Wrap = wdFindStop
        
        Do While .Execute
            Dim latexStr As String
            latexStr = rng.Text
            latexStr = Mid(latexStr, 2, Len(latexStr) - 2)
            
            Dim MyData As Object
            Set MyData = CreateObject("New:{1C3B4210-F441-11CE-B9EA-00AA006B1A69}")
            MyData.SetText latexStr
            MyData.PutInClipboard
            
            rng.Text = ""
            Dim eq As InlineShape
            Set eq = rng.InlineShapes.AddOLEObject(ClassType:="Equation.3", FileName:="", LinkToFile:=False, DisplayAsIcon:=False)
            
            rng.Collapse wdCollapseEnd
        Loop
    End With
    MsgBox "Đã hoàn tất chuyển đổi!", vbInformation, "Thành công"
End Sub

' --- CODE KIỂM TRA CẬP NHẬT TỪ GITHUB PAGES ---
Sub CheckForUpdates()
    Dim currentVersion As String
    Dim remoteVersion As String
    Dim updateUrl As String
    Dim downloadUrl As String
    Dim localPath As String
    
    currentVersion = "1.0" ' Phải khớp với số phiên bản trong file version.txt
    
    ' Đường dẫn trực tiếp trỏ đến website GitHub Pages của bạn
    updateUrl = "https://saobay.github.io/word-addin/version.txt"
    downloadUrl = "https://saobay.github.io/word-addin/SmartTestTools.dotm"
    
    On Error GoTo ErrorHandler
    
    Dim http As Object
    Set http = CreateObject("MSXML2.XMLHTTP")
    http.Open "GET", updateUrl & "?t=" & Timer, False
    http.Send
    
    If http.Status = 200 Then
        remoteVersion = Trim(http.responseText)
        
        If remoteVersion > currentVersion Then
            If MsgBox("Đã có phiên bản mới (" & remoteVersion & "). Bạn có muốn cập nhật không?", vbYesNo + vbInformation, "Cập nhật Add-in") = vbYes Then
                Dim adoStream As Object
                http.Open "GET", downloadUrl & "?t=" & Timer, False
                http.Send
                
                If http.Status = 200 Then
                    localPath = Application.StartupPath & "\SmartTestTools.dotm"
                    Set adoStream = CreateObject("ADODB.Stream")
                    adoStream.Open
                    adoStream.Type = 1
                    adoStream.Write http.responseBody
                    adoStream.SaveToFile localPath, 2
                    adoStream.Close
                    
                    MsgBox "Cập nhật thành công! Vui lòng khởi động lại Word.", vbInformation, "Thành công"
                End If
            End If
        Else
            MsgBox "Bạn đang dùng phiên bản mới nhất (" & currentVersion & ").", vbInformation, "Thông báo"
        End If
    Else
        MsgBox "Không thể kiểm tra cập nhật từ máy chủ.", vbCritical, "Lỗi"
    End If
    Exit Sub
ErrorHandler:
    MsgBox "Lỗi kết nối: " & Err.Description, vbCritical, "Lỗi"
End Sub
