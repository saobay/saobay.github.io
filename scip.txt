function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var type = (e && e.parameter && e.parameter.type) ? e.parameter.type : "Diem"; 
    
    if (type === "ThuMuc") {
      var sheet = ss.getSheetByName("ThuMuc");
      
      if (!sheet || sheet.getLastRow() <= 1) {
        if (!sheet) sheet = ss.insertSheet("ThuMuc");
        sheet.clear();
        sheet.appendRow(["FolderID", "FolderName", "ParentID", "ThoiGian"]);
        
        syncDriveFolderToSheet(ROOT_FOLDER_ID, "ROOT", sheet);
      }
      
      var rows = sheet.getDataRange().getValues();
      var result = [];
      for (var i = 1; i < rows.length; i++) {
        result.push({
          "FolderID": String(rows[i][0]),
          "FolderName": String(rows[i][1]),
          "ParentID": String(rows[i][2]),
          "ThoiGian": String(rows[i][3])
        });
      }
      return responseJSON(result);
    }
    
    var sheet = ss.getSheetByName(type);
    if (!sheet || sheet.getLastRow() <= 1) {
      return responseJSON([]);
    }
    
    var rows = sheet.getDataRange().getValues();
    var headers = rows[0];
    var result = [];
    
    for (var i = 1; i < rows.length; i++) {
      var obj = {};
      for (var j = 0; j < headers.length; j++) {
        obj[headers[j]] = rows[i][j];
      }
      result.push(obj);
    }
    
    return responseJSON(result);
  } catch(err) {
    return responseJSON([]);
  }
}

function processAndUploadImagesToDrive(html, titlePrefix, folder) {
  if (!html) return "";
  
  var base64ImageRegex = /src=["'](data:image\/(png|jpeg|jpg|gif|svg\+xml);base64,([^"']+))["']/gi;
  var match;
  var imgIndex = 1;
  
  while ((match = base64ImageRegex.exec(html)) !== null) {
    try {
      var fullMatch = match[1];
      var ext = match[2] === 'svg+xml' ? 'svg' : match[2];
      var mimeType = match[2] === 'svg+xml' ? 'image/svg+xml' : 'image/' + match[2];
      var base64Data = match[3];
      
      var bytes = Utilities.base64Decode(base64Data);
      var blob = Utilities.newBlob(bytes, mimeType, titlePrefix + "_img_" + imgIndex + "." + ext);
      
      var file = folder.createFile(blob);
      try {
        file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      } catch(e) {}
      
      var driveDirectUrl = "https://drive.google.com/uc?export=view&id=" + file.getId();
      
      html = html.replace(fullMatch, driveDirectUrl);
      imgIndex++;
    } catch (err) {
      Logger.log("Lỗi xử lý ảnh: " + err.toString());
    }
  }
  
  return html;
}

function pushFileToGitHub(filePath, content, commitMessage) {
  if (!GITHUB_TOKEN || !GITHUB_REPO) {
    throw new Error("Thiếu GITHUB_TOKEN hoặc GITHUB_REPO trong cấu hình.");
  }
  
  var url = "https://api.github.com/repos/" + GITHUB_REPO + "/contents/" + filePath;
  
  // 1. Kiểm tra file đã tồn tại trên GitHub hay chưa để lấy SHA
  var sha = "";
  var checkOptions = {
    "method": "GET",
    "headers": {
      "Authorization": "token " + GITHUB_TOKEN,
      "Accept": "application/vnd.github.v3+json"
    },
    "muteHttpExceptions": true
  };
  
  var checkRes = UrlFetchApp.fetch(url, checkOptions);
  var checkCode = checkRes.getResponseCode();
  
  if (checkCode === 200) {
    var fileInfo = JSON.parse(checkRes.getContentText());
    sha = fileInfo.sha;
  }
  
  // 2. Mã hóa UTF-8 sang Base64
  var encodedContent = Utilities.base64Encode(content, Utilities.Charset.UTF_8);
  
  var payload = {
    "message": commitMessage,
    "content": encodedContent
  };
  if (sha) payload.sha = sha;
  
  var pushOptions = {
    "method": "PUT",
    "headers": {
      "Authorization": "token " + GITHUB_TOKEN,
      "Content-Type": "application/json",
      "Accept": "application/vnd.github.v3+json"
    },
    "payload": JSON.stringify(payload),
    "muteHttpExceptions": true
  };
  
  var pushRes = UrlFetchApp.fetch(url, pushOptions);
  var resCode = pushRes.getResponseCode();
  var resText = pushRes.getContentText();
  
  Logger.log("--> Push GitHub File: " + filePath);
  Logger.log("--> Push Status Code: " + resCode);
  Logger.log("--> Push Response Text: " + resText);
  
  if (resCode === 200 || resCode === 201) {
    return JSON.parse(resText);
  } else {
    var errObj = {};
    try { errObj = JSON.parse(resText); } catch(e){}
    var msg = errObj.message || resText;
    throw new Error("GitHub API trả về mã lỗi [" + resCode + "]: " + msg);
  }
}

function syncDriveFolderToSheet(parentFolderId, parentTag, sheet) {
  try {
    var parentFolder = DriveApp.getFolderById(parentFolderId);
    var subFolders = parentFolder.getFolders();
    
    while (subFolders.hasNext()) {
      var sub = subFolders.next();
      var subId = sub.getId();
      sheet.appendRow([subId, sub.getName(), parentTag === "ROOT" ? parentFolderId : parentTag, new Date().toLocaleString()]);
      
      syncDriveFolderToSheet(subId, subId, sheet);
    }
  } catch(e) {}
}

function responseJSON(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
         .setMimeType(ContentService.MimeType.JSON);
}
